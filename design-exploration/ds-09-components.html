// React Native — AffinityBar util
function getAffinityTier(pct: number) {
  if (pct >= 81) return { token: 'aff5', color: '#3A9E7A' };
  if (pct >= 61) return { token: 'aff4', color: '#6B9B7A' };
  if (pct >= 41) return { token: 'aff3', color: '#C89B5C' };
  if (pct >= 21) return { token: 'aff2', color: '#D07777' };
  return { token: 'aff1', color: '#B85C5C' };
}
